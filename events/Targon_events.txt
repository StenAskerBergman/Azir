    picture = GFX_event_TAR_pick_leader
    is_triggered_only = yes

    trigger ={
        # Origin Country
        tag = TAR
    }

    # Leona Remain's - Default
    option = {
        name = TAR_LEONA_NAME
        trigger = {
            has_character = TAR_leona
        }
        set_politics = {
            ruling_party = theocracy
            elections_allowed = no
        }
        promote_character = TAR_leona
        TAR_leona = {
            add_country_leader_trait = {
                trait = TAR_purists
            }
        }
        country_event = { id = targon.9 }

        ai_chance = { factor = 45 }
    }

    option = {
        name = TAR_DIANA_NAME
        trigger = { has_character = TAR_diana }
        set_country_flag = TAR_diana_mandate
        remove_ideas = purist_policy
        remove_ideas = TAR_zealous_crusader
        set_politics = { ruling_party = oligarchy elections_allowed = no }
        add_popularity = { ideology = oligarchy popularity = 0.15 }
        promote_character = TAR_diana
        add_power_balance_value = { id = TAR_balance_of_power value = 0.25 tooltip_side = TAR_lunari_side }
        ai_chance = { factor = 40 }
    }

    # Rahvun Leader Path
    option = {
        name = TAR_RAHVUN_NAME
        trigger = {
            has_character = TAR_rahvun
        }
        custom_effect_tooltip = targon.20.b.tt

        if = {
            limit = {
                has_character = TAR_leona
            }
            TAR_leona = {
                remove_country_leader_role = {
                    ideology = solari
                }
            }
        }
        add_popularity = {
            ideology = theocracy
            popularity = 0.25
        }
        set_politics = {
            ruling_party = theocracy
            elections_allowed = no
        }
        promote_character = TAR_rahvun
        set_cosmetic_tag = TAR_solari
        ai_chance = {
            factor = 5
        }

        country_event = { id = targon.9 }
    }

    option = {
        name = TAR_PANTHEON_NAME
        ai_chance = { factor = 35 }
        trigger = {
            has_character = TAR_pantheon
        }
        add_popularity = {
            ideology = absolutists
            popularity = 0.25
        }
        set_politics = {
            ruling_party = absolutists
            elections_allowed = no
        }
        promote_character = TAR_pantheon
        set_country_flag = TAR_atreus_mandate
        set_country_flag = TAR_reformation_path
        country_event = { id = TAR_reformers.1 }
    }

    option = {
