country_event = {
	id = magerebellion.3
	title = magerebellion.3.t
	desc = magerebellion.3.d
	picture = GFX_report_event_demons
	
	is_triggered_only = yes
	
	option = { #We must beat back this evil!
		name = magerebellion.3.a
		ai_chance = { factor = 100 }
		if = {
		limit = {
       			 NOT = { has_country_flag = evelynn }
       			 NOT = { has_country_flag = nocturne }
       			 NOT = { has_country_flag = fiddle }
				 NOT = { has_country_flag = found_demons }
		}
		random_owned_state = {
            		add_dynamic_modifier = { 
				modifier = demon_attack 
			}
       		 }
		add_ideas = MREevelynnidea
		set_country_flag = evelynn
		activate_mission = MREevelynnFound
		if = {
			limit = {
				has_country_flag = found_demons
			}
			add_stability = -0.05
			add_political_power = 25
			add_days_mission_timeout  = {
   				 mission = MREevelynnFound
   				 days = -90
			}
		}else = {
			add_stability = -0.15
			add_war_support = -0.10
			add_political_power = -75
			
		}
		set_country_flag = found_demons
		}
	}
}
