# Both parties retain their own choices; AI alone uses these route weights.
country_event = {
    id = nazumah.401
    title = nazumah.401.t
    desc = nazumah.401.d
    picture = GFX_report_event_generic_parliament
    is_triggered_only = yes
    trigger = { tag = NAZ is_subject = yes }
    option = {
        name = nazumah.401.a
        ai_chance = {
            base = 80
            modifier = {
                factor = 0
                overlord = { is_ai = no has_completed_focus = SHU_puppet_nazumah }
            }
        }
        if = {
            limit = { is_subject = yes }
            overlord = { country_event = { id = nazumah.402 } }
        }
    }
    option = {
        name = nazumah.401.b
        ai_chance = {
            base = 20
            modifier = {
                factor = 4
                is_historical_focus_on = yes
                NOT = { has_completed_focus = NAZ_Foe_of_Shurima }
            }
        }
    }
}
