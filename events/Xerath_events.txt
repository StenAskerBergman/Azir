country_event = {
	id = xerath.1
	title = xerath.1.t # shurima.azir.1.t
	desc = xerath.1.d # shurima.azir.1.d
	picture = GFX_report_event_empty_throne
	trigger = { NOT = { has_country_flag = xer_at_war_with_shu } }
	fire_only_once = yes
	is_triggered_only = yes

	immediate = {
		custom_effect_tooltip = xerath.1.tt

		hidden_effect = {
			set_global_flag = SHU_XER_civil_war_trigger
			every_country = {
				limit = {
					OR = {
						tag = XER
						tag = SHU
					}
				}
				scoped_play_song = Miya_Izumi-Shurima-Ashes_of_the_Crescent_2
			}
		}
	}


	option = { # Breaking point!
	name = xerath.1.a
	ai_chance = { factor = 100 }

	set_country_flag = xer_at_war_with_shu
}
}

# Usurp the Throne
country_event = {
	id = xerath.4
	title = xerath.4.t # xerath.4.t
	desc = xerath.4.d # xerath.4.d
	picture = GFX_report_event_empty_throne
	trigger = { NOT = { has_country_flag = xer_throne_usurped } }
	fire_only_once = yes
	is_triggered_only = yes

	immediate = {
		custom_effect_tooltip = xerath.4.tt

		hidden_effect = {
			set_global_flag = xer_usurpation_initiated
		}
	}

	# Option 1: Direct Assault
	option = {
		name = xerath.4.a
		ai_chance = { factor = 50 }
