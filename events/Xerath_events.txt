# SHU proposal uses sender flags. NAZ writes the response in XER scope.
# XER consumes that flag immediately and owns the peace operation.
add_namespace = XER_regional

country_event = {
    id = XER_regional.1
    hidden = yes
    is_triggered_only = yes
    trigger = {
        tag = XER
        OR = {
            AND = {
                has_country_flag = XER_shurima_offer_pending
                FROM = {
                    tag = SHU
                    OR = { has_country_flag = SHU_XER_armistice_accept has_country_flag = SHU_XER_armistice_refuse }
                }
            }
            AND = {
                has_country_flag = XER_nazumah_offer_pending
                FROM = { tag = NAZ }
                OR = { has_country_flag = XER_NAZ_armistice_accepted has_country_flag = XER_NAZ_armistice_refused }
            }
        }
    }
    immediate = {
        if = {
            limit = { FROM = { tag = SHU } }
            clr_country_flag = XER_shurima_offer_pending
            if = {
                limit = {
                    FROM = {
                        has_country_flag = SHU_XER_armistice_accept
                        NOT = { has_country_flag = SHU_XER_armistice_refuse }
                    }
                    XER_regional_shurima_war = yes
                    has_country_flag = XER_frontier_mandates
                }
                white_peace = { tag = SHU }
                if = {
                    limit = { NOT = { has_war_with = SHU } }
                    set_country_flag = XER_shurima_armistice
                    if = {
                        limit = { has_country_flag = XER_command_break }
                        set_truce = { target = SHU days = 90 }
                    }
                    country_event = { id = XER_regional.3 }
                }
                else = { country_event = { id = XER_regional.5 } }
            }
            else_if = {
                limit = { FROM = { has_country_flag = SHU_XER_armistice_refuse } }
                set_country_flag = { flag = XER_shurima_offer_cooldown days = 90 value = 1 }
                country_event = { id = XER_regional.4 }
            }
            else = { country_event = { id = XER_regional.5 } }
            FROM = {
                clr_country_flag = SHU_XER_armistice_accept
                clr_country_flag = SHU_XER_armistice_refuse
            }
        }
        else_if = {
            limit = { FROM = { tag = NAZ } }
            clr_country_flag = XER_nazumah_offer_pending
            if = {
                limit = {
                    has_country_flag = XER_NAZ_armistice_accepted
                    NOT = { has_country_flag = XER_NAZ_armistice_refused }
                    XER_regional_nazumah_war = yes
                    has_country_flag = XER_frontier_mandates
                    NOT = { has_country_flag = XER_command_break }
                }
                white_peace = { tag = NAZ }
                if = {
                    limit = { NOT = { has_war_with = NAZ } }
                    set_country_flag = XER_nazumah_armistice
                    NAZ = { set_country_flag = NAZ_XER_armistice_confirmed }
                    country_event = { id = XER_regional.3 }
                }
                else = { country_event = { id = XER_regional.5 } }
            }
            else_if = {
                limit = { has_country_flag = XER_NAZ_armistice_refused }
                set_country_flag = { flag = XER_offer_nazumah_armistice_cooldown days = 120 value = 1 }
                country_event = { id = XER_regional.4 }
            }
            else = { country_event = { id = XER_regional.5 } }
            clr_country_flag = XER_NAZ_armistice_accepted
            clr_country_flag = XER_NAZ_armistice_refused
        }
    }
}
