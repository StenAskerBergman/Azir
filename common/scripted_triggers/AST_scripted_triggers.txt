	if = {
		limit = { NOT = { has_idea = limited_conscription } }
		custom_trigger_tooltip = {
			tooltip = AST_conscription_level_allowed_tt
			check_variable = { AST_conscription_level_allowed > 0 }
		}
	}
}

	FROM = { has_country_flag = AST_instigated_communist_civil_war }
}

AST_overlord_eng_is_non_monarchy_or_fascist = {
	OR = {
		AST = { is_subject = no }
		ENG = {
			NOT = {
				has_government = neutrality
			}
			OR = {
				NOT = {
					has_idea = george_v
					has_idea = ENG_king_edward
					has_idea = ENG_king_edward_fascism
					has_idea = ENG_george_vi
				}
				has_government = fascism
			}
		}
	}
}
