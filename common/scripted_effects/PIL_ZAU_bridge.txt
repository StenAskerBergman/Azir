PIL_initialize_bop = {
    if = {
        limit = { NOT = { has_power_balance = { id = PIL_bop } } }
        set_power_balance = { id = PIL_bop set_value = 0 }
        if = {
            limit = {
                is_ai = yes
                NOT = { has_country_flag = PIL_ai_prefers_compact }
                NOT = { has_country_flag = PIL_ai_prefers_control }
            }
            random_list = {
                55 = { set_country_flag = PIL_ai_prefers_compact }
                45 = { set_country_flag = PIL_ai_prefers_control }
            }
        }
    }
}
PIL_set_cooperative = {
    set_power_balance = { id = PIL_bop set_value = -0.65 }
}

    set_power_balance = { id = PIL_bop set_value = 0.65 }
}

PIL_ZAU_lock_hardening = {
    if = {
        limit = {
            PIL_ZAU_bridge_live = yes
            PIL_ZAU_bridge_unresolved = yes
            PLT = { PIL_bridge_coercive = yes }
            ZAU = { ZAU_bridge_coercive = yes }
        }
        set_global_flag = PIL_ZAU_hardened
        PLT = { add_ideas = PIL_freight_disruption country_event = { id = PIL_ZAU_events.26 } }
        ZAU = { add_ideas = ZAU_freight_disruption country_event = { id = PIL_ZAU_events.22 } }
    }
}

    }
}

# The charter only benefits a working route. Restore it if the same approach
# states return to their signatories and the cities are at peace again.
PIL_update_joint_freight = {
    if = {
        limit = {
            PIL_ZAU_joint_freight_operating = yes
            NOT = { has_idea = PIL_joint_freight }
        }
        add_ideas = PIL_joint_freight
    }
    else_if = {
        limit = {
            has_idea = PIL_joint_freight
            NOT = { PIL_ZAU_joint_freight_operating = yes }
        }
        remove_ideas = PIL_joint_freight
    }
}
