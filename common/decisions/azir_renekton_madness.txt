XER_renekton_madness_category = {
    XER_hear_the_tomb = {
        icon = generic_prepare_civil_war
        cost = 0
        fire_only_once = yes

        visible = {
            NOT = { has_country_flag = XER_renekton_madness_unveiled }
            any_controlled_state = { NOT = { is_core_of = ROOT } }
        }

        complete_effect = {
            set_country_flag = XER_renekton_madness_unveiled
            set_country_flag = XER_renekton_madness_active
            set_variable = { XER_renekton_tainted_states = 0 }
            activate_mission = XER_renekton_last_days
            mark_focus_tree_layout_dirty = yes
            country_event = { id = azir_renekton.1 }
        }

        # The hidden route can end the ruler's life. Leave the initial choice to the player.
        ai_will_do = { factor = 0 }
    }

    XER_renekton_last_days = {
        icon = generic_prepare_civil_war
        days_mission_timeout = 120
        is_good = no
        fire_only_once = yes
        activation = { always = no }
        available = { always = no }
        cancel_trigger = { NOT = { XER_renekton_crisis_active = yes } }
        cancel_effect = { clr_country_flag = XER_renekton_madness_active }

        timeout_effect = {
            if = {
                limit = { XER_renekton_crisis_active = yes }
                clr_country_flag = XER_renekton_madness_active
                set_country_flag = XER_renekton_died_in_crisis
                kill_country_leader = yes
                country_event = { id = azir_renekton.6 }
                if = {
                    limit = { SHU = { has_character = SHU_nasus } }
                    SHU = { country_event = { id = azir_renekton.6 } }
                }
            }
        }
    }

    XER_spread_renekton_madness = {
        icon = generic_prepare_civil_war
        cost = 25
        state_target = yes
        target_array = controlled_states
        target_root_trigger = { XER_renekton_can_spread_madness = yes }
        target_trigger = {
            FROM = {
                is_controlled_by = ROOT
                NOT = { is_core_of = ROOT }
                NOT = { has_state_flag = XER_renekton_state_tainted }
            }
        }
        on_map_mode = map_and_decisions_view

        complete_effect = {
            FROM = {
                set_state_flag = XER_renekton_state_tainted
                add_dynamic_modifier = { modifier = XER_renekton_tainted_land }
            }
            hidden_effect = { add_to_variable = { XER_renekton_tainted_states = 1 } }
        }

        ai_will_do = { factor = 1 }
    }

    XER_answer_the_vigil = {
        icon = generic_prepare_civil_war
        cost = 0
        fire_only_once = yes
        visible = {
            XER_renekton_crisis_active = yes
            SHU = {
                SHU_renekton_treatment_active = yes
                check_variable = { SHU_nasus_vigils > 2 }
            }
        }
        complete_effect = {
            hidden_effect = {
                SHU = {
                    set_country_flag = SHU_nasus_corrupted
                    remove_mission = SHU_renekton_treatment
                    SHU_nasus = {
                        set_portraits = {
                            civilian = {
                                large = "GFX_portrait_SHU_nasus_corrupted_large"
                                small = "GFX_portrait_SHU_nasus_corrupted_small"
                            }
                        }
                    }
                    add_ideas = SHU_nasus_fractured
                    country_event = { id = azir_renekton.7 }
                }
            }
        }
        ai_will_do = { factor = 1 }
    }
}
